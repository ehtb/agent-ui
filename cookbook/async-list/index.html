A list loaded from the server, with loading, failure and retry

Cookbook recipe: A list loaded from the server, with loading, failure and retry

Edit this page on GitHub↗

Data that lives on the server. A resources entry runs the request and writes only through its lifecycle actions; a retry flips a watched key. The page says when it is ready (ready) so the server waits for the data.

pages/async-list/page.js

page.js
// Recipe: a list loaded from the server, with loading, failure and retry.
// The namespace that needs the data declares the request (`resources`) next to its state and actions. The request never runs in an
// action: the plugin runs it and reports through the three lifecycle actions, the only ones allowed to write what the resource `owns`.
// A retry is not a special case: it changes a key the resource watches, and the request runs again.
import { withResources } from '@ehtb/agent-ui/resources';

const books = {
  state: { list: [], status: 'idle', error: '', attempt: 0 },
  derive: {
    count: (get) => `${get('list').length} books`,
  },
  actions: {
    // Not while the page is ready: the browser repeats the request after the server rendered the finished page, and that must not turn it back into a loading one.
    loading: { args: {}, run: (store) => { if (store.get('status') !== 'ready') store.set('status', 'loading'); store.set('error', ''); } },
    loaded: { args: { data: 'json' }, run: (store, { data }) => { store.set('list', data); store.set('status', 'ready'); } },
    failed: { args: { message: 'string' }, run: (store, { message }) => { store.set('status', 'failed'); store.set('error', message); } },
    retry: { args: {}, run: (store) => store.set('attempt', store.get('attempt') + 1) },
  },
  resources: [
    {
      watch: ['attempt'],
      owns: ['list', 'status', 'error'],
      request: () => ({ url: '/books.json' }),
      onStart: 'loading',
      onSuccess: 'loaded',
      onError: 'failed',
    },
  ],
};

export const page = {
  head: { title: 'Books - cookbook' },
  namespaces: { books },
  plugins: [withResources],
  // The server renders the page once the request has finished, successfully or not; a failed load is a finished state with a message.
  ready: (state) => state.books.status === 'ready' || state.books.status === 'failed',
  status: (state) => (state.books.status === 'failed' ? 503 : 200),
};

pages/async-list/view.js

view.js
import { act, button, each, equals, h1, item, li, main, p, state, ul, when } from '@ehtb/agent-ui/spec';

export const view = main(
  { id: 'books-page' },
  h1(null, 'Books'),
  when(equals(state('books.status'), 'loading'), p({ id: 'loading', role: 'status' }, 'Loading...')),
  when(equals(state('books.status'), 'failed'), [p({ id: 'error', role: 'alert' }, state('books.error')), button({ id: 'retry', on: { click: act('books.retry') } }, 'Try again')]),
  when(equals(state('books.status'), 'ready'), [p({ id: 'count' }, state('books.count')), ul({ id: 'books' }, each('books.list', 'id', li({ id: 'book' }, item('title'), ' (', item('year'), ')')))]),
);

server-fetch.mjs

server-fetch.mjs
// Where the recipes get data while a page is rendered on the server: the files in public/, the same bytes the browser fetches.
import { readFile } from 'node:fs/promises';
import { dirname, resolve, sep } from 'node:path';
import { fileURLToPath } from 'node:url';

const publicDirectory = resolve(dirname(fileURLToPath(import.meta.url)), 'public');

export const fetchFunction = async (url) => {
  try {
    const file = resolve(publicDirectory, `.${decodeURIComponent(new URL(url, 'http://localhost').pathname)}`);
    if (!file.startsWith(publicDirectory + sep)) return new Response(null, { status: 404 });
    return new Response(await readFile(file, 'utf8'), { headers: { 'content-type': 'application/json' } });
  } catch {
    return new Response(null, { status: 404 });
  }
};

build.config.mjs

build.config.mjs
// The build configuration of the cookbook, a project like any other: `agent-ui check` in this folder renders every recipe.
import { dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { frameworkBuildOptions } from '@ehtb/agent-ui/build/presets.mjs';
import { fetchFunction } from './server-fetch.mjs';

const projectRoot = dirname(fileURLToPath(import.meta.url));

export default ({ dev = false, validator } = {}) => ({
  ...frameworkBuildOptions({ dev, validator, projectRoot }),
  publicDir: 'public',
  fetchFunction, // answers the data requests of pages rendered on the server (the async-list recipe)
});
