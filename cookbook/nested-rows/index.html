Rows inside rows that read the outer row

Cookbook recipe: Rows inside rows that read the outer row

Edit this page on GitHub↗

A list of groups, each with a list of items, where the inner template needs a field of the outer row (an id for a handler, a name for a label). Name the outer row with as, read it with row(...); no copying of outer fields into the inner items.

2 filesCopy them in reading order and rename the namespaces.

pages/nested-rows/page.js

page.js
// Recipe: rows inside rows that read the outer row.
// An order has lines. A line's remove button needs the id of its order, and a line's label shows the customer. In the view the outer
// `each` says `as: 'order'`, and anything inside it, at any depth, reads that row with row('order.id'); item(...) is still the
// innermost row (the line). The data stays as it is: no outer fields copied into every line in `derive`.
const orders = {
  state: {
    list: [
      { id: 1, customer: 'Ada', lines: [{ id: 10, sku: 'pen' }, { id: 11, sku: 'ink' }] },
      { id: 2, customer: 'Bo', lines: [{ id: 20, sku: 'pad' }] },
    ],
  },
  shapes: { list: { fields: { id: 'number', customer: 'string', lines: 'json' } } }, // so row('order.custmer') is caught at mount
  actions: {
    removeLine: {
      args: { orderId: 'number', lineId: 'number' },
      run: (store, { orderId, lineId }) =>
        store.set('list', store.get('list').map((order) => (order.id === orderId ? { ...order, lines: order.lines.filter((line) => line.id !== lineId) } : order))),
    },
  },
};

export const page = { head: { title: 'Orders - cookbook' }, namespaces: { orders } };

pages/nested-rows/view.js

view.js
import { act, button, each, h2, item, join, li, main, row, section, span, ul } from '@ehtb/agent-ui/spec';

const lineRow = li(
  { id: 'line' },
  span({ id: 'line-label' }, join(row('order.customer'), ': ', item('sku'))),
  button({ id: 'line-remove', 'aria-label': join('Remove ', item('sku'), ' from order ', row('order.id')), on: { click: act('orders.removeLine', { orderId: row('order.id'), lineId: item('id') }) } }, 'x'),
);

const orderSection = section({ id: 'order' }, h2({ id: 'order-title' }, 'Order ', item('id')), ul({ id: 'lines' }, each(item('lines'), 'id', lineRow)));

export const view = main({ id: 'orders-page' }, each('orders.list', 'id', orderSection, { as: 'order' }));
