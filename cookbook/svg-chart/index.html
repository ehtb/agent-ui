A bar chart in SVG: bars from data, a gradient, axis labels and a tooltip on each bar

Cookbook recipe: A bar chart in SVG: bars from data, a gradient, axis labels and a tooltip on each bar

Edit this page on GitHub↗

A chart that fits in the spec. The position and height of each bar and the label above the picked one are derived, one display row per bar, and the template reads them; text draws the labels, linearGradient (referred to by domId) fills the bars, a title inside a shape is its tooltip, and the svg is role: 'img' with a summary bound to state, with the same numbers in buttons that also pick a bar from the keyboard. A chart with zoom, pointer-following tooltips or thousands of marks is a library drawn by a plugin (recipe draw-into-element).

pages/svg-chart/page.js

page.js
// Recipe: a bar chart in SVG, with a gradient, axis labels and a tooltip on every bar.
// The chart is spec like any other markup: what is drawn from data (the position and the height of each bar, the label above the picked
// one) is computed in `derive`, one display row per bar, and the template reads it. A chart that needs zoom, pointer-following tooltips or
// thousands of marks is a library drawn by a plugin (recipe `draw-into-element`); a chart that fits in the spec needs no plugin.
const WIDTH = 220;
const PLOT_HEIGHT = 80;

const sales = {
  state: { months: [{ id: 'jan', label: 'Jan', value: 12 }, { id: 'feb', label: 'Feb', value: 19 }, { id: 'mar', label: 'Mar', value: 8 }, { id: 'apr', label: 'Apr', value: 24 }], pickedId: null },
  derive: {
    // One row per bar, in a 220 wide box with the highest month at the top of an 80 high plot. Equal rows keep their reference,
    // so changing the picked month rewrites two bars and no more.
    bars: (get) => {
      const months = get('months');
      const highest = Math.max(...months.map((month) => month.value), 1);
      const slot = WIDTH / months.length;
      return months.map((month, index) => {
        const height = (month.value / highest) * PLOT_HEIGHT;
        return {
          id: month.id,
          label: month.label,
          x: index * slot + slot * 0.15,
          width: slot * 0.7,
          y: PLOT_HEIGHT - height,
          height,
          center: index * slot + slot / 2,
          valueY: PLOT_HEIGHT - height - 3,
          tooltip: `${month.label}: ${month.value}`,
          picked: month.id === get('pickedId'),
        };
      });
    },
    summary: (get) => {
      const months = get('months');
      const best = months.reduce((top, month) => (month.value > top.value ? month : top), months[0]);
      return `Sales by month, ${months.length} months, the most in ${best.label} with ${best.value}`;
    },
  },
  actions: {
    pick: { args: { id: 'string' }, run: (store, { id }) => store.set('pickedId', store.get('pickedId') === id ? null : id) },
  },
};

export const page = { head: { title: 'Chart - cookbook' }, namespaces: { sales } };

pages/svg-chart/view.js

view.js
import { act, button, defs, each, g, h1, item, li, linearGradient, main, rect, state, stop, svg, text, title, ul, when } from '@ehtb/agent-ui/spec';

// A drawing that says something: an image with a label that a screen reader reads instead of the shapes (so the same numbers are in the
// buttons below, which are also how a keyboard picks a bar). The label is bound to state, so it follows the data.
export const view = main(
  { id: 'chart-page' },
  h1(null, 'Sales'),
  svg(
    { id: 'chart', viewBox: '0 0 220 100', role: 'img', 'aria-label': state('sales.summary') },
    // The gradient is referred to by its HTML id (domId); `id` is the spec's own address of the element.
    defs(null, linearGradient({ id: 'bar-gradient', domId: 'bar-fill', x1: 0, y1: 0, x2: 0, y2: 1 }, stop({ offset: '0%', 'stop-color': 'currentColor' }), stop({ offset: '100%', 'stop-color': 'currentColor', 'stop-opacity': 0.35 }))),
    each(
      'sales.bars',
      'id',
      g(
        { classes: { picked: item('picked') } },
        // A title inside a shape is the browser's tooltip for it.
        rect({ x: item('x'), y: item('y'), width: item('width'), height: item('height'), fill: 'url(#bar-fill)' }, title(null, item('tooltip'))),
        text({ x: item('center'), y: 93, 'text-anchor': 'middle', 'font-size': 7 }, item('label')),
        when(item('picked'), text({ x: item('center'), y: item('valueY'), 'text-anchor': 'middle', 'font-size': 7, 'font-weight': 'bold' }, item('tooltip'))),
      ),
    ),
  ),
  ul(
    { id: 'months' },
    each('sales.bars', 'id', li(null, button({ id: 'month', 'aria-pressed': item('picked'), on: { click: act('sales.pick', { id: item('id') }) } }, item('tooltip')))),
  ),
);

/* The chart takes the colour of the text around it (the gradient uses currentColor); a picked bar is darker. */
#chart-page svg { width: 100%; max-width: 28rem; color: #3b6fd4; }
#chart-page g rect { opacity: 0.75; }
#chart-page g.picked rect { opacity: 1; }
#chart-page text { fill: #333; font-family: system-ui, sans-serif; }
#months { display: flex; gap: 0.5rem; padding: 0; list-style: none; }
#months button[aria-pressed="true"] { font-weight: 700; }
