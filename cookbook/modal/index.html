A dialog that opens, closes on Escape and keeps the focus until it closes

Cookbook recipe: A dialog that opens, closes on Escape and keeps the focus until it closes

Edit this page on GitHub↗

Anything that sits above the page until it is dismissed. open is state; the dialog exists only while it is open (when); the content behind it is inert; focus.js focuses the dialog, wraps Tab and Shift+Tab inside it, and gives the focus back to the control that opened it.

3 filesCopy them in reading order and rename the namespaces.

pages/modal/page.js

page.js
// Recipe: a dialog that opens, closes on Escape and keeps the focus until it closes.
// Whether it is open is state. The dialog is in the page only while it is open (`when`), the page behind it is `inert`, and
// the plugin focus.js moves the focus in, wraps Tab inside the dialog and gives the focus back to the button that opened it.
// Opening and closing are idempotent.
import { withDialogFocus } from './focus.js';

const dialog = {
  state: { open: false, confirmed: 0 },
  actions: {
    show: { args: {}, run: (store) => store.set('open', true) },
    close: { args: {}, run: (store) => store.set('open', false) },
    confirm: {
      args: {},
      run: (store) => {
        if (!store.get('open')) return;
        store.set('confirmed', store.get('confirmed') + 1);
        store.set('open', false);
      },
    },
  },
};

export const page = { head: { title: 'Dialog - cookbook' }, namespaces: { dialog }, plugins: [withDialogFocus] };

pages/modal/view.js

view.js
import { act, button, dialog, div, h2, main, p, state, when } from '@ehtb/agent-ui/spec';

// The page behind the dialog is inert while it is open: it looks the same, but neither a pointer nor the Tab key reaches it.
// The dialog itself is a sibling of that content, not a child, so `inert` never covers the dialog.
export const view = main(
  { id: 'modal-page' },
  div(
    { id: 'page-content', inert: state('dialog.open') },
    button({ id: 'open', on: { click: act('dialog.show') } }, 'Delete everything'),
    p({ id: 'confirmed' }, 'Confirmed ', state('dialog.confirmed'), ' times'),
  ),
  when(
    state('dialog.open'),
    dialog(
      { id: 'dialog', open: true, 'aria-modal': true, 'aria-labelledby': 'dialog-title', tabindex: '-1', on: { 'keydown.Escape': act('dialog.close') } },
      h2({ id: 'dialog-title', domId: 'dialog-title' }, 'Are you sure?'),
      button({ id: 'confirm', on: { click: act('dialog.confirm') } }, 'Yes, delete'),
      button({ id: 'cancel', on: { click: act('dialog.close') } }, 'Cancel'),
    ),
  ),
);

pages/modal/focus.js

focus.js
// Recipe: focus stays in an open dialog and returns to the control that opened it.
//
// The plugin owns the focus, not the state: `dialog.open` is still the only thing that says whether the dialog exists. It
// watches the spec element with id "dialog" through api.observeElements (the same hook mountElement is built on), focuses the
// dialog itself (it has tabindex="-1" so a screen reader reads its label), wraps Tab and Shift+Tab inside it, and gives the
// focus back to the element that had it before the dialog appeared. The page marks the rest of the content `inert`, so the
// browser's own focus order cannot reach it either.
//
// `isServerRendering` is true while the page is rendered on the server: there is no focus there, only markup.
export const withDialogFocus = ({ isServerRendering, win = globalThis }) => (api) => {
  if (isServerRendering) return;
  let dialog = null;
  let returnFocusTo = null;
  let onKeydown = null;
  let restoreTimer = null;

  // The controls of this dialog. The recipe gives every one an id, so [data-id] is enough and works in the test DOM too.
  const focusable = (element) =>
    [...element.querySelectorAll('[data-id]')].filter(
      (node) => ['button', 'a', 'input', 'select', 'textarea'].includes(node.localName) && !node.disabled && node.getAttribute('tabindex') !== '-1',
    );

  const forgetDialog = () => {
    if (dialog && onKeydown) dialog.removeEventListener?.('keydown', onKeydown);
    dialog = null;
    onKeydown = null;
  };

  const cancelRestore = () => {
    if (restoreTimer !== null) win.clearTimeout?.(restoreTimer);
    restoreTimer = null;
  };

  // Later, after the render that closed the dialog has also removed `inert` from the page behind it.
  const restoreFocus = () => {
    restoreTimer = null;
    if (dialog) return; // the dialog was opened again before the timer ran
    const target = returnFocusTo;
    returnFocusTo = null;
    if (target && (typeof target.ownerDocument?.contains !== 'function' || target.ownerDocument.contains(target))) target.focus?.();
  };

  const stopObserving = api.observeElements({
    added: (element) => {
      if (dialog || element.getAttribute('data-id') !== 'dialog') return;
      cancelRestore(); // a pending restore of an earlier dialog must not steal this one's focus
      dialog = element;
      returnFocusTo = element.ownerDocument?.activeElement ?? null;
      element.focus?.(); // the dialog itself: its aria-labelledby names it, and Tab starts at its first control
      onKeydown = (event) => {
        if (event.key !== 'Tab') return;
        const items = focusable(element);
        if (!items.length) return void event.preventDefault?.();
        const active = element.ownerDocument?.activeElement;
        const index = items.indexOf(active);
        if (event.shiftKey && index <= 0) {
          event.preventDefault?.();
          items[items.length - 1].focus();
        } else if (!event.shiftKey && (index === -1 || index === items.length - 1)) {
          event.preventDefault?.();
          items[0].focus();
        }
      };
      element.addEventListener?.('keydown', onKeydown);
    },
    removed: (node) => {
      if (node !== dialog) return;
      forgetDialog();
      // The rest of the page is inert until this same render has finished; wait for it before focusing the old control.
      cancelRestore();
      restoreTimer = win.setTimeout(restoreFocus, 0);
    },
  });

  api.onDestroy(() => {
    stopObserving();
    cancelRestore();
    forgetDialog();
    returnFocusTo = null; // the trigger is going away with the page: there is nowhere to put the focus back
  });
};
