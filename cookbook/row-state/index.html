State that belongs to each row of a list (open, selected, a note)

Cookbook recipe: State that belongs to each row of a list (open, selected, a note)

Edit this page on GitHub↗

Every row of a list has its own open/closed or draft text. Declare rowState with the defaults, write with store.setRow(id, ...), read with rowState(...) in the row. Do not invent a keyed map by hand.

pages/row-state/page.js

page.js
// Recipe: state that belongs to each row of a list.
// Each person can be opened to show a note field. Whether a row is open and what its note says are per-row state: declared once with
// their defaults (`rowState`), read in the row with rowState('details.open'), written by the namespace's own actions with
// store.setRow(id, { ... }). The state is one key, `details.rowState`, holding only the rows that differ from the defaults, so it is
// plain data: logged, replayed, sent from the server. A row's state stays when its item leaves the list until the owner forgets it:
// removing a person calls `details.forget`.
const details = {
  rowState: { open: false, note: '' },
  actions: {
    toggle: { args: { id: 'number' }, run: (store, { id }) => store.setRow(id, { open: !store.getRow(id).open }) },
    write: { args: { id: 'number', note: 'string' }, run: (store, { id, note }) => store.setRow(id, { note }) },
    forget: { args: { id: 'number' }, run: (store, { id }) => store.clearRow(id) },
  },
};

const people = {
  state: {
    items: [
      { id: 1, name: 'Ada' },
      { id: 2, name: 'Bo' },
      { id: 3, name: 'Cy' },
    ],
  },
  actions: {
    remove: {
      args: { id: 'number' },
      calls: ['details.forget'],
      run: (store, { id }, call) => {
        store.set('items', store.get('items').filter((person) => person.id !== id));
        call('details.forget', { id });
      },
    },
  },
};

export const page = { head: { title: 'People - cookbook' }, namespaces: { people, details } };

pages/row-state/view.js

view.js
import { act, button, each, event, h1, item, li, main, rowState, span, textarea, ul, when } from '@ehtb/agent-ui/spec';

const personRow = li(
  { id: 'person', classes: { open: rowState('details.open') } },
  span({ id: 'person-name' }, item('name')),
  button({ id: 'person-toggle', on: { click: act('details.toggle', { id: item('id') }) } }, 'Notes'),
  button({ id: 'person-remove', on: { click: act('people.remove', { id: item('id') }) } }, 'Remove'),
  when(rowState('details.open'), textarea({ id: 'person-note', 'aria-label': 'Note', value: rowState('details.note'), on: { input: act('details.write', { id: item('id'), note: event('target.value') }) } })),
);

export const view = main({ id: 'people-page' }, h1(null, 'People'), ul({ id: 'people' }, each('people.items', 'id', personRow)));
