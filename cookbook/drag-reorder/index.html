A list reordered by dragging a row, or from the keyboard

Cookbook recipe: A list reordered by dragging a row, or from the keyboard

Edit this page on GitHub↗

Drag and drop. There is no drag-and-drop API (no dragstart, drop or dataTransfer): a grip button picks a row up on pointerdown, entering another row moves it there, releasing drops it, and every step is a logged action. follow (plugin withFollow) keeps the row under the pointer (the coordinates never enter the state) and transition slides the others aside (plugin withMoves); the page lists both plugins. Escape and a cancelled pointer put the row back (the start position is state). A touch pointer never reports the rows it crosses, so the keyboard (Enter, the arrow keys) is both the touch path and the accessible one: build it every time.

pages/drag-reorder/page.js

page.js
import { withFollow } from '@ehtb/agent-ui/follow';
import { withMoves } from '@ehtb/agent-ui/moves';

// Recipe: a list reordered by dragging a row, or from the keyboard.
// There is no drag-and-drop API here (no dragstart, drop or dataTransfer): the gesture is the bubbling pointer events the grammar has,
// and every step is a logged action, so a drag replays like anything else. Pressing a grip picks the row up (`press`), moving over another
// row puts the picked-up one in its place (`over`), releasing ends the drag (`release`). The pointer's coordinates are not state:
// `follow` (see view.js) keeps the picked-up row under the pointer, and `transition` slides the other rows aside: both are plugins the
// page lists, `withFollow` (@ehtb/agent-ui/follow) and `withMoves` (@ehtb/agent-ui/moves). A touch pointer is captured by the grip, so
// `over` never fires for it: the keyboard (Enter picks up and puts down, the arrow keys `move`, Escape `cancel`s) is the touch path and
// the accessible path, and a list that has no keyboard path is not finished.
const movedItem = (items, from, to) => {
  const next = items.slice();
  next.splice(to, 0, ...next.splice(from, 1));
  return next;
};
const indexOfId = (items, id) => items.findIndex((task) => task.id === id);

// Picking up, putting down and cancelling all end the drag the same way.
const endDrag = (store) => {
  store.set('draggingId', null);
  store.set('startIndex', -1);
};

// Moves the picked-up row to the place of row `id` (nothing happens when it is already there or nothing is picked up).
const moveDraggedTo = (store, id) => {
  const items = store.get('items');
  const from = indexOfId(items, store.get('draggingId'));
  const to = indexOfId(items, id);
  if (from >= 0 && to >= 0 && from !== to) store.set('items', movedItem(items, from, to));
};

const tasks = {
  // startIndex is where the picked-up row began, so Escape can put it back.
  state: {
    items: [
      { id: 1, title: 'Draft the outline' },
      { id: 2, title: 'Write the tests' },
      { id: 3, title: 'Ship it' },
    ],
    draggingId: null,
    startIndex: -1,
  },
  derive: {
    hint: (get) => (get('draggingId') === null ? 'Drag a grip over the other rows, or focus a grip and press Enter, then use the up and down arrows. Escape puts it back.' : 'Moving. Drag over a row, or use the arrows and Enter. Escape puts it back where it started.'),
  },
  actions: {
    // Pointer down on a grip, or Enter on it, in one action, so a grip is always a meaningful target: with nothing picked up this row is
    // picked up; on the picked-up row it is put down; on any other row the picked-up row moves there and is put down.
    press: {
      args: { id: 'number' },
      run: (store, { id }) => {
        if (store.get('draggingId') === null) {
          store.set('draggingId', id);
          store.set('startIndex', indexOfId(store.get('items'), id));
          return;
        }
        moveDraggedTo(store, id);
        endDrag(store);
      },
    },
    // The pointer crosses a row while dragging: the picked-up row takes that row's place right away.
    over: { args: { id: 'number' }, run: (store, { id }) => moveDraggedTo(store, id) },
    // The pointer is released: the drag ends where the row is. Decided by the release, not by where the row ended up (a row dragged out
    // and back to its first place is still dropped). It is idempotent: with nothing picked up it does nothing.
    release: {
      args: {},
      run: (store) => {
        if (store.get('draggingId') !== null) endDrag(store);
      },
    },
    // Escape, or a cancelled pointer (the browser took it, for example a touch scroll): put the row back where it started.
    cancel: {
      args: {},
      run: (store) => {
        const items = store.get('items');
        const from = indexOfId(items, store.get('draggingId'));
        const to = store.get('startIndex');
        if (from >= 0 && to >= 0 && from !== to) store.set('items', movedItem(items, from, to));
        endDrag(store);
      },
    },
    // The arrow keys move the picked-up row by one place.
    move: {
      args: { delta: 'number' },
      run: (store, { delta }) => {
        const items = store.get('items');
        const from = indexOfId(items, store.get('draggingId'));
        const to = from + delta;
        if (from >= 0 && to >= 0 && to < items.length) store.set('items', movedItem(items, from, to));
      },
    },
  },
};

export const page = { head: { title: 'Reorder - cookbook' }, namespaces: { tasks }, plugins: [withFollow, withMoves] };

pages/drag-reorder/view.js

view.js
import { act, button, each, equals, h1, item, join, li, main, p, span, state, ul } from '@ehtb/agent-ui/spec';

const isDragged = equals(state('tasks.draggingId'), item('id'));

export const view = main(
  // A release or a cancelled pointer anywhere on the page ends the drag, not only inside the list. With nothing picked up they do nothing.
  { id: 'reorder-page', on: { pointerup: act('tasks.release'), pointercancel: act('tasks.cancel') } },
  h1(null, 'Tasks'),
  ul(
    // Escape anywhere in the list puts a picked-up row back.
    { id: 'tasks', on: { 'keydown.Escape': act('tasks.cancel') } },
    each(
      'tasks.items',
      'id',
      li(
        // `transition` slides the other rows aside as the list reorders (plugin withMoves); `follow` keeps the picked-up row under the pointer.
        { id: 'task', transition: 150, follow: isDragged, classes: { dragging: isDragged }, on: { pointerover: act('tasks.over', { id: item('id') }) } },
        button(
          {
            id: 'grip',
            type: 'button',
            'aria-label': join('Reorder ', item('title')),
            'aria-pressed': isDragged,
            focus: isDragged, // the grip keeps the focus while its row moves, so the arrow keys keep working
            on: {
              // preventDefault stops text selection and the browser's own drag start.
              pointerdown: act('tasks.press', { id: item('id') }, { preventDefault: true }),
              'keydown.Enter': act('tasks.press', { id: item('id') }, { preventDefault: true }),
              'keydown.ArrowUp': act('tasks.move', { delta: -1 }, { preventDefault: true }),
              'keydown.ArrowDown': act('tasks.move', { delta: 1 }, { preventDefault: true }),
            },
          },
          '⣿',
        ),
        span(null, item('title')),
      ),
    ),
  ),
  p({ id: 'hint' }, state('tasks.hint')),
);

pages/drag-reorder/page.css

page.css
/* The picked-up row follows the pointer: --follow-x and --follow-y are written by the runtime while `follow` is true. It must not
   catch the pointer itself, or the rows beneath it never receive pointerover. */
#tasks { margin: 0; padding: 0; list-style: none; display: grid; gap: 0.5rem; user-select: none; }
#tasks li { display: flex; align-items: center; gap: 0.5rem; padding: 0.5rem 0.75rem; border: 1px solid #ccc; background: #fff; }
/* touch-action: none, so a touch drag on the grip does not scroll the page. */
#tasks li button { cursor: grab; touch-action: none; border: 0; background: transparent; font: inherit; }
#tasks li.dragging { transform: translate(var(--follow-x, 0px), var(--follow-y, 0px)); pointer-events: none; opacity: 0.85; box-shadow: 0 4px 12px rgb(0 0 0 / 0.2); }
