A form whose errors, and whether it can be submitted, follow from its fields

Cookbook recipe: A form whose errors, and whether it can be submitted, follow from its fields

Edit this page on GitHub↗

Validation is derived from the fields, never stored: emailError, passwordError and valid are derive, the errors only show after the first attempt (attempted), and submit finishes only when valid.

2 filesCopy them in reading order and rename the namespaces.

pages/form-validation/page.js

page.js
// Recipe: a form whose errors, and whether it can be submitted, follow from its fields.
// The fields are state. The errors and `valid` are derived from them, so they cannot disagree with the fields. Errors are shown
// only after the first attempt to submit (`attempted`), not while the visitor is still typing the first character.
const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

const signup = {
  state: { email: '', password: '', attempted: false, done: false },
  derive: {
    emailError: (get) => (EMAIL.test(get('email')) ? '' : 'Enter an address like name@example.com'),
    passwordError: (get) => (get('password').length >= 8 ? '' : 'Use at least 8 characters'),
    valid: (get) => get('emailError') === '' && get('passwordError') === '',
    // What the view shows: an error only once the visitor has tried to submit.
    shownEmailError: (get) => (get('attempted') ? get('emailError') : ''),
    shownPasswordError: (get) => (get('attempted') ? get('passwordError') : ''),
  },
  actions: {
    setEmail: { args: { value: 'string' }, run: (store, { value }) => store.set('email', value) },
    setPassword: { args: { value: 'string' }, run: (store, { value }) => store.set('password', value) },
    submit: {
      args: {},
      run: (store) => {
        store.set('attempted', true);
        if (store.get('valid')) store.set('done', true);
      },
    },
  },
};

export const page = { head: { title: 'Sign up - cookbook' }, namespaces: { signup } };

pages/form-validation/view.js

view.js
import { act, button, div, event, form, h1, input, label, length, greaterThan, main, p, state, when } from '@ehtb/agent-ui/spec';

const field = (name, labelText, type, setter) => [
  label({ for: `${name}-field` }, labelText),
  input({ id: name, domId: `${name}-field`, type, 'aria-label': labelText, value: state(`signup.${name}`), on: { input: act(setter, { value: event('target.value') }) } }),
  when(greaterThan(length(state(`signup.shown${name[0].toUpperCase()}${name.slice(1)}Error`)), 0), p({ id: `${name}-error`, role: 'alert' }, state(`signup.shown${name[0].toUpperCase()}${name.slice(1)}Error`))),
];

export const view = main(
  { id: 'signup-page' },
  h1(null, 'Sign up'),
  when(
    state('signup.done'),
    p({ id: 'welcome', role: 'status' }, 'Welcome aboard.'),
    form(
      { id: 'signup-form', on: { submit: act('signup.submit') } },
      div(null, ...field('email', 'Email', 'email', 'signup.setEmail')),
      div(null, ...field('password', 'Password', 'password', 'signup.setPassword')),
      button({ id: 'submit', type: 'submit' }, 'Create account'),
    ),
  ),
);
