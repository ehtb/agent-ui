A list you can add to, edit in place, tick and remove from

Cookbook recipe: A list you can add to, edit in place, tick and remove from

Edit this page on GitHub↗

Any list of records the visitor changes. The list, the draft being typed and the row being edited are three namespaces; the one that changes another goes through calls. Every item has its own id (the list's key); agent-ui check fails on a duplicate.

2 filesCopy them in reading order and rename the namespaces.

pages/list-edit/page.js

page.js
// Recipe: a list you can add to, edit in place, tick and remove from.
// Three namespaces, one concern each. `notes` owns the list; `composer` owns the text being typed; `editing` owns which row is
// being edited and its text. When one of them has to change another (adding clears the draft, saving renames a note) it lists
// the other's action in `calls` and the owner does the writing.
const notes = {
  state: { items: [], nextId: 1 },
  derive: {
    remaining: (get) => get('items').filter((note) => !note.done).length,
  },
  actions: {
    add: {
      args: { text: 'string' },
      run: (store, { text }) => {
        const id = store.get('nextId');
        store.set('nextId', id + 1);
        store.set('items', [...store.get('items'), { id, text, done: false }]);
      },
    },
    toggle: { args: { id: 'number' }, run: (store, { id }) => store.set('items', store.get('items').map((note) => (note.id === id ? { ...note, done: !note.done } : note))) },
    rename: { args: { id: 'number', text: 'string' }, run: (store, { id, text }) => store.set('items', store.get('items').map((note) => (note.id === id ? { ...note, text } : note))) },
    remove: { args: { id: 'number' }, run: (store, { id }) => store.set('items', store.get('items').filter((note) => note.id !== id)) },
  },
};

const composer = {
  state: { draft: '' },
  actions: {
    setDraft: { args: { value: 'string' }, run: (store, { value }) => store.set('draft', value) },
    // An empty draft adds nothing, and pressing the button twice adds one note: the draft is cleared by the first.
    submit: {
      args: {},
      calls: ['notes.add'],
      run: (store, args, call) => {
        const text = store.get('draft').trim();
        if (!text) return;
        call('notes.add', { text });
        store.set('draft', '');
      },
    },
  },
};

const editing = {
  state: { id: null, draft: '' },
  actions: {
    start: {
      args: { id: 'number', text: 'string' },
      run: (store, { id, text }) => {
        store.set('id', id);
        store.set('draft', text);
      },
    },
    setDraft: { args: { value: 'string' }, run: (store, { value }) => store.set('draft', value) },
    // Saving and cancelling are safe to repeat: the second call finds nothing being edited (blur follows Enter in a browser).
    save: {
      args: {},
      calls: ['notes.rename'],
      run: (store, args, call) => {
        const id = store.get('id');
        if (id === null) return;
        const text = store.get('draft').trim();
        if (text) call('notes.rename', { id, text });
        store.set('id', null);
        store.set('draft', '');
      },
    },
    cancel: {
      args: {},
      run: (store) => {
        store.set('id', null);
        store.set('draft', '');
      },
    },
  },
};

export const page = { head: { title: 'Notes - cookbook' }, namespaces: { notes, composer, editing } };

pages/list-edit/view.js

view.js
import { act, button, div, each, equals, event, form, h1, input, item, li, main, not, span, state, ul, when } from '@ehtb/agent-ui/spec';

const isEditing = equals(state('editing.id'), item('id'));

const noteRow = li(
  { id: 'note', classes: { done: item('done') } },
  input({ id: 'note-done', type: 'checkbox', 'aria-label': 'Done', checked: item('done'), on: { change: act('notes.toggle', { id: item('id') }) } }),
  when(
    not(isEditing),
    span({ id: 'note-text', on: { dblclick: act('editing.start', { id: item('id'), text: item('text') }) } }, item('text')),
    input({
      id: 'note-edit',
      'aria-label': 'Edit note',
      value: state('editing.draft'),
      focus: true,
      on: { input: act('editing.setDraft', { value: event('target.value') }), 'keydown.Enter': act('editing.save'), 'keydown.Escape': act('editing.cancel'), focusout: act('editing.save') },
    }),
  ),
  button({ id: 'note-remove', 'aria-label': 'Remove', on: { click: act('notes.remove', { id: item('id') }) } }, 'x'),
);

export const view = main(
  { id: 'notes-page' },
  h1(null, 'Notes'),
  form(
    { id: 'composer', on: { submit: act('composer.submit') } },
    input({ id: 'draft', 'aria-label': 'New note', value: state('composer.draft'), on: { input: act('composer.setDraft', { value: event('target.value') }) } }),
    button({ id: 'add', type: 'submit' }, 'Add'),
  ),
  ul({ id: 'notes' }, each('notes.items', 'id', noteRow)),
  div({ id: 'remaining' }, state('notes.remaining'), ' left'),
);
