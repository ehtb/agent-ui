Icons and a small chart in SVG

Cookbook recipe: Icons and a small chart in SVG

Edit this page on GitHub↗

An icon next to text is aria-hidden, sized with 1em and coloured with currentColor; an icon used many times comes from a sprite in public/ with use; a drawing that says something is role: 'img' with an aria-label. What is drawn from data (the stars, the points of a line) is a derived value. A large or interactive chart belongs to a library drawn by a plugin (recipe draw-into-element).

3 filesCopy them in reading order and rename the namespaces.

pages/svg-icons/page.js

page.js
// Recipe: icons and a small chart in SVG.
// SVG is a closed set of tags (svg, g, path, circle, ellipse, rect, line, polyline, polygon, use): its shapes stand inside an svg, and
// an svg either says what it shows (role: 'img' with aria-label) or is decoration next to text that says it ('aria-hidden': 'true').
// What is drawn from data is computed in `derive`, like any other value: the stars to show, and the points of the line.
const rating = {
  state: { value: 3 },
  derive: {
    // Five stars, each on or off. Equal values keep their reference, so only the stars that changed are written again.
    stars: (get) => [1, 2, 3, 4, 5].map((id) => ({ id, on: id <= get('value') })),
  },
  actions: {
    set: { args: { value: 'number' }, run: (store, { value }) => store.set('value', value) },
  },
};

const visits = {
  state: { perDay: [3, 5, 2, 8, 6, 9, 7] },
  derive: {
    // The points of a polyline in a 60 by 20 box: one step of 10 per day, the highest day at the top.
    points: (get) => {
      const perDay = get('perDay');
      const highest = Math.max(...perDay, 1);
      return perDay.map((count, day) => `${day * 10},${20 - (count / highest) * 20}`).join(' ');
    },
    summary: (get) => `Visits this week, from ${get('perDay')[0]} to ${get('perDay').at(-1)} a day`,
  },
  actions: {
    add: { args: { count: 'number' }, run: (store, { count }) => store.set('perDay', [...store.get('perDay').slice(1), count]) },
  },
};

export const page = { head: { title: 'Icons - cookbook' }, namespaces: { rating, visits } };

pages/svg-icons/view.js

view.js
import { act, button, each, h1, item, join, li, main, p, path, polyline, section, span, state, svg, ul, use } from '@ehtb/agent-ui/spec';

// A decorative icon next to the text that says it: hidden from assistive technology, sized and coloured by the text around it.
const checkIcon = svg({ viewBox: '0 0 24 24', width: '1em', height: '1em', 'aria-hidden': 'true', fill: 'none', stroke: 'currentColor', 'stroke-width': 2 }, path({ d: 'M5 12l5 5L20 7' }));

// An icon from a sprite file in public/: one request for every icon of the site. The href is a path of the site (the base path is added).
const star = svg({ viewBox: '0 0 24 24', width: 24, height: 24, 'aria-hidden': 'true' }, use({ href: '/icons.svg#star' }));

export const view = main(
  { id: 'icons-page' },
  h1(null, 'Icons'),
  button({ id: 'save' }, checkIcon, ' Save'),
  section(
    { id: 'rating', 'aria-label': 'Rating' },
    ul({ id: 'stars' }, each('rating.stars', 'id', li(null, button({ id: 'star', classes: { on: item('on') }, 'aria-label': join('Rate ', item('id')), 'aria-pressed': item('on'), on: { click: act('rating.set', { value: item('id') }) } }, star)))),
    p({ id: 'rating-text' }, state('rating.value'), ' of 5'),
  ),
  // A drawing that says something: an image with a label, which a screen reader reads instead of the shapes.
  svg({ id: 'sparkline', viewBox: '0 0 60 20', width: 120, height: 40, role: 'img', 'aria-label': state('visits.summary'), fill: 'none', stroke: 'currentColor' }, polyline({ id: 'sparkline-line', points: state('visits.points') })),
  span({ id: 'sparkline-legend' }, 'Last seven days'),
);
