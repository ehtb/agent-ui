A header that stays while the page changes (soft navigation)

Cookbook recipe: A header that stays while the page changes (soft navigation)

Edit this page on GitHub↗

Something that must survive a link click (a header, a cart count, a player). A shell is an app of its own around the page; a link marked soft: true swaps only the page.

6 filesCopy them in reading order and rename the namespaces.

pages/soft-shell/page.js

page.js
// The page "/soft-shell": behavior only. It lists the shell it lives in, and the namespace the shell reads.
import { shell, visit } from './shell.js';

export const page = { head: { title: 'First page - cookbook' }, namespaces: { visit }, shell };

pages/soft-shell/view.js

view.js
import { a, act, button, h1, section, state } from '@ehtb/agent-ui/spec';
import { shell } from './shell-view.js';

export const view = section(
  { id: 'page-content' },
  h1({ id: 'page-title' }, 'First page'),
  button({ id: 'bump', on: { click: act('visit.bump') } }, 'Bump: ', state('visit.count')),
  a({ id: 'other-link', href: '/soft-shell/other', soft: true }, 'Go to the second page'),
);
export { shell };

pages/soft-shell/shell.js

shell.js
// Recipe: a header that stays while the page changes. Behavior of the shell, and the namespace both pages share.
// The shell is an app of its own around the page. `frame.clicks` lives in the shell, so it survives every soft link; `visit.count`
// lives in the page, which is replaced, so it starts again on each page. The shell shows the page's count (`reads`, one way).
import { defineShell } from '@ehtb/agent-ui/shell';
import { withNavigation } from '@ehtb/agent-ui/navigation';

export const visit = {
  state: { count: 0 },
  actions: { bump: { args: {}, run: (store) => store.set('count', store.get('count') + 1) } },
};

const frame = {
  state: { clicks: 0, pageCount: 0 },
  actions: {
    click: { args: {}, run: (store) => store.set('clicks', store.get('clicks') + 1) },
    setPageCount: { args: { count: 'number' }, run: (store, { count }) => store.set('pageCount', count) },
  },
};

export const shell = defineShell({
  name: 'main',
  namespaces: { frame },
  plugins: [withNavigation], // makes `soft: true` links and route.go work
  reads: [{ watch: ['visit.count'], action: 'frame.setPageCount', args: { count: 'visit.count' } }],
});

pages/soft-shell/shell-view.js

shell-view.js
// The content of the shell. The view.js of every page that uses the shell exports it; the build ships it once, as data.
import { a, act, button, equals, footer, header, main, nav, pageSlot, small, state } from '@ehtb/agent-ui/spec';

export const shell = {
  name: 'main',
  view: main(
    { id: 'frame' },
    header(
      { id: 'frame-header' },
      nav(
        { id: 'frame-nav' },
        a({ id: 'nav-first', href: '/soft-shell', soft: true, classes: { active: equals(state('route.path'), '/soft-shell') } }, 'First'),
        a({ id: 'nav-second', href: '/soft-shell/other', soft: true, classes: { active: equals(state('route.path'), '/soft-shell/other') } }, 'Second'),
      ),
      button({ id: 'frame-click', on: { click: act('frame.click') } }, 'Shell clicks: ', state('frame.clicks')),
      small({ id: 'frame-page-count' }, 'Page count: ', state('frame.pageCount')),
    ),
    pageSlot(), // the page goes here
    footer({ id: 'frame-footer' }, small(null, 'The header above is not replaced when the page changes.')),
  ),
};
