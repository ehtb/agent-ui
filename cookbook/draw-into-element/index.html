A chart, an editor or a map drawn by a plugin into one element

Cookbook recipe: A chart, an editor or a map drawn by a plugin into one element

Edit this page on GitHub↗

Anything that owns its own DOM. The spec has an empty element with an explicit id; a plugin claims it with mountElement, draws into it, reports what happens with api.dispatch, and cleans up when the element leaves the page.

3 filesCopy them in reading order and rename the namespaces.

pages/draw-into-element/page.js

page.js
// Recipe: a chart, an editor or a map drawn by a plugin into one element.
// The page has no idea how the chart draws. It owns the data (`sales.series`) and the selection; the plugin (bars.js) draws the data
// and reports a click as `sales.select`. The element in view.js is empty and has an explicit id, which is how the plugin finds it.
import { withBars } from './bars.js';

const sales = {
  state: { series: [3, 5, 2], selected: -1 },
  actions: {
    add: { args: { value: 'number' }, run: (store, { value }) => store.set('series', [...store.get('series'), value]) },
    select: { args: { index: 'number' }, run: (store, { index }) => store.set('selected', index) },
  },
};

export const page = { head: { title: 'Chart - cookbook' }, namespaces: { sales }, plugins: [withBars] };

pages/draw-into-element/view.js

view.js
import { act, button, div, h1, main, p, state, when, greaterThan, length } from '@ehtb/agent-ui/spec';

export const view = main(
  { id: 'chart-page' },
  h1(null, 'Sales'),
  div({ id: 'bars', role: 'img', 'aria-label': 'Sales by month' }), // empty on purpose: the plugin draws here
  button({ id: 'add', on: { click: act('sales.add', { value: 4 }) } }, 'Add a month'),
  p({ id: 'selected' }, 'Selected: ', state('sales.selected')),
  when(greaterThan(length(state('sales.series')), 5), p(null, 'That is a lot of months.')),
);

pages/draw-into-element/bars.js

bars.js
// A plugin that draws into an element: here, a bar chart made of text, standing for a charting library. Everything such a plugin
// does is in this file: claim the element by its id (mountElement), draw into it, follow the state it shows (api.watch), report what
// the visitor does in it (api.dispatch), and give everything back when the element leaves the page.
//
// The page lists it like any plugin: plugins: [withBars]. It never touches state itself; it reads through api.watch and writes
// through api.dispatch, so what happens in the chart is a logged action like a click anywhere else.
import { mountElement } from '@ehtb/agent-ui/mount-element';

const drawBars = (element, series) => {
  element.replaceChildren();
  series.forEach((value, index) => {
    const bar = element.ownerDocument.createElement('div');
    bar.setAttribute('data-bar', String(index));
    bar.append(element.ownerDocument.createTextNode('#'.repeat(value)));
    element.append(bar);
  });
};

// A page lists it as `plugins: [withBars]`: the page hands every plugin its environment (win, fetchFunction), and this one needs none.
export const withBars = () => (api) =>
  mountElement(api, 'bars', (element) => {
    const stop = api.watch(['sales.series'], (values) => drawBars(element, values['sales.series']));
    // The element belongs to the plugin: its own listener, removed with it. A click on a bar is an action, like any other.
    const onClick = (event) => {
      const index = event.target.getAttribute?.('data-bar');
      if (index !== null && index !== undefined && !api.replaying) api.dispatch('sales.select', { index: Number(index) });
    };
    element.addEventListener('click', onClick);
    return () => {
      stop();
      element.removeEventListener('click', onClick);
      element.replaceChildren();
    };
  });
